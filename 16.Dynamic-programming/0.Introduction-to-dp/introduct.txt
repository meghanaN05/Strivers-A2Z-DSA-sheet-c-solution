Dynamic Programming (DP) is an optimization technique used to solve complex problems by breaking them down into smaller, overlapping subproblems. Instead of recomputing the solution to identical subproblems multiple times, DP solves each subproblem just once and stores its result for future use.
When to use Dynamic Programming?A problem can be solved using DP if it satisfies two main properties:Overlapping Subproblems: The same subproblems are solved repeatedly (e.g., computing Fibonacci(3) multiple times while calculating Fibonacci(5)).Optimal Substructure: The optimal solution to the overall problem can be constructed using the optimal solutions of its subproblems (e.g., Fibonacci(n) = Fibonacci(n-1) + Fibonacci(n-2)).Standard Example: Fibonacci NumbersTo understand all four approaches, let's find the nth Fibonacci number, defined as:$F(0) = 0$$F(1) = 1$$F(n) = F(n-1) + F(n-2)
#include <iostream>
using namespace std;

// Time Complexity: O(2^n) - Exponential due to duplicate recursive calls
// Space Complexity: O(n)   - Auxiliary stack space for recursion
int fibonacci(int n) {
    if (n <= 1) return n;
    return fibonacci(n - 1) + fibonacci(n - 2);
}

int main() {
    int n = 6;
    cout << "Fibonacci(" << n << ") = " << fibonacci(n) << endl;
    return 0;
}
#include <iostream>
#include <vector>
using namespace std;

// Time Complexity: O(n) - Each state from 0 to n is computed only once
// Space Complexity: O(n) [Recursion Stack] + O(n) [Memoization Array] = O(n)
int fibonacciMemo(int n, vector<int>& dp) {
    if (n <= 1) return n;
    
    // Return cached result if already computed
    if (dp[n] != -1) return dp[n];
    
    // Store and return result
    return dp[n] = fibonacciMemo(n - 1, dp) + fibonacciMemo(n - 2, dp);
}

int main() {
    int n = 6;
    vector<int> dp(n + 1, -1);
    cout << "Fibonacci(" << n << ") = " << fibonacciMemo(n, dp) << endl;
    return 0;
}
#include <iostream>
#include <vector>
using namespace std;

// Time Complexity: O(n) - Single iterative loop from 2 to n
// Space Complexity: O(n) - Array to store DP states (No recursion stack overhead)
int fibonacciTabulation(int n) {
    if (n <= 1) return n;
    
    vector<int> dp(n + 1, 0);
    dp[0] = 0;
    dp[1] = 1;
    
    for (int i = 2; i <= n; i++) {
        dp[i] = dp[i - 1] + dp[i - 2];
    }
    
    return dp[n];
}

int main() {
    int n = 6;
    cout << "Fibonacci(" << n << ") = " << fibonacciTabulation(n) << endl;
    return 0;
}
#include <iostream>
using namespace std;

// Time Complexity: O(n) - Single loop from 2 to n
// Space Complexity: O(1) - Constant extra space
int fibonacciOptimized(int n) {
    if (n <= 1) return n;
    
    int prev2 = 0; // Represents F(i-2)
    int prev1 = 1; // Represents F(i-1)
    int curr = 0;  // Represents F(i)
    
    for (int i = 2; i <= n; i++) {
        curr = prev1 + prev2;
        prev2 = prev1;
        prev1 = curr;
    }
    
    return prev1;
}

int main() {
    int n = 6;
    cout << "Fibonacci(" << n << ") = " << fibonacciOptimized(n) << endl;
    return 0;
}
